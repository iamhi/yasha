# Yasha CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a CLI tool (`yasha`) that updates all `package.json` dependencies to their latest versions and performs a clean reinstall.

**Architecture:** Single `bin/yasha.js` file with a `#!/usr/bin/env node` shebang. Reads `package.json` from `cwd`, shells out to `npm view <pkg> version` for each dep using `execFileSync` (no shell injection risk), rewrites versions, then deletes `node_modules/` and `package-lock.json` before running `npm install`. Zero runtime dependencies — Node.js stdlib only.

**Tech Stack:** Node.js (stdlib: `node:fs`, `node:child_process`, `node:path`), `node:test` for tests.

### Task 2: Implement the CLI

**Files:**
- Create: `bin/yasha.js`

- [ ] **Step 1: Create `bin/` directory and `bin/yasha.js`**

```js
#!/usr/bin/env node

import { readFileSync, writeFileSync, rmSync, existsSync } from 'node:fs'
import { execFileSync } from 'node:child_process'
import { join } from 'node:path'

const cwd = process.cwd()
const pkgPath = join(cwd, 'package.json')

if (!existsSync(pkgPath)) {
  console.error('Error: no package.json found in current directory')
  process.exit(1)
}

const pkg = JSON.parse(readFileSync(pkgPath, 'utf8'))

const depGroups = ['dependencies', 'devDependencies']

for (const group of depGroups) {
  if (!pkg[group]) continue
  for (const name of Object.keys(pkg[group])) {
    try {
      const latest = execFileSync('npm', ['view', name, 'version'], { encoding: 'utf8' }).trim()
      pkg[group][name] = latest
    } catch {
      console.warn(`Warning: could not resolve latest version for ${name}, skipping`)
    }
  }
}

writeFileSync(pkgPath, JSON.stringify(pkg, null, 2) + '\n', 'utf8')

const nodeModulesPath = join(cwd, 'node_modules')
if (existsSync(nodeModulesPath)) {
  rmSync(nodeModulesPath, { recursive: true, force: true })
}

const lockPath = join(cwd, 'package-lock.json')
if (existsSync(lockPath)) {
  rmSync(lockPath)
}

execFileSync('npm', ['install'], { stdio: 'inherit', cwd })
```

# Fake npm: "npm view <pkg> version" always returns "9.9.9"

- [ ] **Step 2: Create `test/yasha.test.js`**

```js
import { test } from 'node:test'
import assert from 'node:assert/strict'
import { mkdtempSync, writeFileSync, readFileSync, mkdirSync, existsSync } from 'node:fs'
import { execFileSync } from 'node:child_process'
import { join, resolve } from 'node:path'
import { tmpdir } from 'node:os'

const yashaPath = resolve('bin/yasha.js')
const mockNpmDir = resolve('test')

function runYasha(cwd) {
  return execFileSync('node', [yashaPath], {
    cwd,
    encoding: 'utf8',
    env: { ...process.env, PATH: `${mockNpmDir}:${process.env.PATH}` }
  })
}

test('updates dependencies and devDependencies to latest', () => {
  const dir = mkdtempSync(join(tmpdir(), 'yasha-test-'))
  writeFileSync(join(dir, 'package.json'), JSON.stringify({
    name: 'test-pkg',
    dependencies: { express: '^4.0.0' },
    devDependencies: { jest: '^27.0.0' }
  }, null, 2))

  runYasha(dir)

  const result = JSON.parse(readFileSync(join(dir, 'package.json'), 'utf8'))
  assert.equal(result.dependencies.express, '9.9.9')
  assert.equal(result.devDependencies.jest, '9.9.9')
})

test('removes node_modules and package-lock.json before installing', () => {
  const dir = mkdtempSync(join(tmpdir(), 'yasha-test-'))
  writeFileSync(join(dir, 'package.json'), JSON.stringify({
    name: 'test-pkg',
    dependencies: { express: '^4.0.0' }
  }, null, 2))

  mkdirSync(join(dir, 'node_modules'))
  writeFileSync(join(dir, 'package-lock.json'), '{}')

  runYasha(dir)

  assert.equal(existsSync(join(dir, 'package-lock.json')), false)
})

test('exits with error when no package.json found', () => {
  const dir = mkdtempSync(join(tmpdir(), 'yasha-test-'))
  assert.throws(
    () => execFileSync('node', [yashaPath], {
      cwd: dir,
      encoding: 'utf8',
      env: { ...process.env, PATH: `${mockNpmDir}:${process.env.PATH}` }
    }),
    (err) => {
      assert.match(err.stderr, /no package\.json found/)
      return true
    }
  )
})

test('preserves other package.json fields', () => {
  const dir = mkdtempSync(join(tmpdir(), 'yasha-test-'))
  writeFileSync(join(dir, 'package.json'), JSON.stringify({
    name: 'my-app',
    version: '2.0.0',
    description: 'hello',
    dependencies: { lodash: '4.0.0' }
  }, null, 2))

  runYasha(dir)

  const result = JSON.parse(readFileSync(join(dir, 'package.json'), 'utf8'))
  assert.equal(result.name, 'my-app')
  assert.equal(result.version, '2.0.0')
  assert.equal(result.description, 'hello')
  assert.equal(result.dependencies.lodash, '9.9.9')
})
```
